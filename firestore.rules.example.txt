rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() {
      return signedIn() && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // Developer submissions: owner can create/read/update own; admin can read/update all
    match /submissions/{id} {
      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow read: if signedIn() && (resource.data.ownerUid == request.auth.uid || isAdmin());
      allow update: if signedIn() && (resource.data.ownerUid == request.auth.uid || isAdmin());
      allow delete: if isAdmin();
    }

    // Approved/published addons visible to everyone (marketplace reads); only admin writes
    match /published/{addonId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Admin markers (create/manage manually in console or by existing admin if you add rule)
    match /admins/{uid} {
      allow read: if signedIn() && request.auth.uid == uid;
      allow write: if false; // set via Firebase console
    }
  }
}
// NOTE: The portal/admin UI also uses MAINTAINER_UIDS/MAINTAINER_EMAILS in firebase-config.js
